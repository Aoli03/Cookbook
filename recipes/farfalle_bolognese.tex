\setRecipeMeta{Farfalle Bolognese}{4}{15 MIN}{35 MIN}{./images/CookbookDefault.jpg}

\begin{recipe}
  \begin{ingredients}
    \ingredientGroup{Pasta}
    \ingredient{400g farfalle pasta}
    \ingredient{Salt, for pasta water}
    \ingredientGroup{Bolognese}
    \ingredient{500g ground beef (or beef/pork mix)}
    \ingredient{1 medium onion, diced}
    \ingredient{2 medium carrots, diced}
    \ingredient{1 red bell pepper, diced}
    \ingredient{3 cloves garlic, minced}
    \ingredient{2 tbsp tomato paste}
    \ingredient{400g crushed tomatoes}
    \ingredient{100ml red wine (optional)}
    \ingredient{2 large handfuls of spinach}
    \ingredient{1 tsp dried oregano}
    \ingredient{1 tsp dried basil}
    \ingredient{Salt and pepper to taste}
    \ingredient{2 tbsp olive oil}
    \ingredient{Parmesan, to serve}
  \end{ingredients}
  \begin{steps}
    \step{Bring a large pot of salted water to a boil. Cook farfalle until al dente according to package instructions. Reserve 1 cup pasta water before draining.}
    \step{Heat olive oil in a large pan over medium heat. Sauté onion, carrots, and bell pepper for 6--8 minutes until softened.}
    \step{Add garlic and cook for 1 minute. Push vegetables to the side, add ground meat, and brown for 5--6 minutes, breaking it up as it cooks.}
    \step{Stir in tomato paste and cook for 2 minutes. Add wine if using and let it reduce by half, about 2 minutes.}
    \step{Add crushed tomatoes, oregano, basil, salt, and pepper. Simmer uncovered on low heat for 15 minutes, stirring occasionally.}
    \step{Stir in spinach and cook until wilted, about 2 minutes. Add drained pasta and toss to combine, loosening with reserved pasta water as needed. Serve with Parmesan.}
  \end{steps}
\end{recipe}
